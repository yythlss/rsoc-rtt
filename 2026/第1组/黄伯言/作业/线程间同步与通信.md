## 练习一：停车场车位管理

### 场景

一个停车场只有 3 个车位，同时有 5 辆车尝试进入。车位属于有限资源，车辆只有获得车位后才能进入。

### 功能要求

1. 创建一个初始值为 3 的计数信号量，表示剩余车位。

2. 创建 5 个车辆线程。

3. 每辆车到达后申请一个车位。

4. 获得车位后打印车辆编号和进入信息。

5. 车辆停留一段时间后离开，并释放车位。

6. 未在 2 秒内获得车位的车辆打印等待超时信息，稍后重新申请。

7. 任意时刻进入停车场的车辆数不能超过 3。
- main(msh)

```
/* ===== MSH 命令主函数 ===== */
static void cmd_parking_demo(int argc, char **argv)
{
    int i;
    char name[16];

    /* 1. 检查是否已在运行 */
    if (parking_task_running) {
        rt_kprintf("Parking demo is already running!\n");
        rt_kprintf("Please wait for it to finish, or reset the system.\n");
        return;
    }

    /* 2. 如果处于"残留"状态（资源已释放但标志未清），先清理 */
    if (parking_sem != RT_NULL) {
        parking_cleanup();
    }

    rt_kprintf("Starting parking demo...\n");

    /* 3. 创建信号量（初始值 3） */
    parking_sem = rt_sem_create("parking_sem", 3, RT_IPC_FLAG_PRIO);
    if (parking_sem == RT_NULL) {
        rt_kprintf("Create semaphore failed!\n");
        return;
    }

    /* 4. 创建 5 个车辆线程 */
    for (i = 0; i < 5; i++) {
        rt_snprintf(name, sizeof(name), "car%d", i + 1);
        car_threads[i] = rt_thread_create(name,
                                          car_thread_entry,
                                          (void *)(size_t)(i + 1),
                                          1024,
                                          20 + i,
                                          10);
        if (car_threads[i] == RT_NULL) {
            rt_kprintf("Create car%d thread failed!\n", i + 1);
            parking_cleanup();
            return;
        }
        rt_thread_startup(car_threads[i]);
        rt_kprintf("Car %d thread created.\n", i + 1);
    }

    /* 5. 创建管理线程（优先级 10，高于车辆线程） */
    manager_thread = rt_thread_create("manager",
                                      manager_thread_entry,
                                      RT_NULL,
                                      1024,
                                      10,
                                      10);
    if (manager_thread == RT_NULL) {
        rt_kprintf("Create manager thread failed!\n");
        parking_cleanup();
        return;
    }
    rt_thread_startup(manager_thread);
    rt_kprintf("Manager thread created.\n");

    /* 6. 设置运行标志 */
    parking_task_running = 1;
    rt_kprintf("Parking demo started with 5 cars and 3 slots.\n");
}

void car_entry(void *parameter)
{
    /* 取出车号（将 void* 转换为 int） */
    int car_id = (int)parameter;

    while(1)
    {
        /* 所有车辆执行相同的逻辑，但打印时使用不同的 car_id */
                rt_kprintf("Car %d: Arrived, applying for a slot...\n", car_id);

                {
                    rt_kprintf("Car %d: ENTERED! Remaining slots: %d\n",
                               car_id, parking_sem->value);

                    int stay_ms = (car_id * 300 + 1000) % 4000 + 1000;
                    rt_thread_mdelay(stay_ms);
    
                    rt_sem_release(parking_sem);
                    rt_kprintf("Car %d: LEFT! Remaining slots: %d\n",
                               car_id, parking_sem->value);
                }
                else
                {
                    rt_kprintf("Car %d: TIMEOUT! No slot available.\n", car_id);
                }
    
                /* 等待一段时间后再次尝试 */
                rt_thread_mdelay(3000);
            }
    }

## 练习二：多窗口售票系统

### 场景

两个售票窗口同时出售同一批车票。系统初始有 20 张票，每次只能由一个窗口修改剩余票数。

### 功能要求

1. 创建两个售票线程，分别代表窗口 A 和窗口 B。
2. 使用一个全局变量保存剩余票数。
3. 每个窗口循环完成以下操作：
   - 检查是否还有余票；
   - 将余票减 1；
   - 记录售出的票号；
   - 打印售票结果。
4. 使用互斥量保护完整的售票过程。
5. 售完后两个线程自动退出。
6. 最终必须满足：
   - 20 张票全部售出；
   - 不出现相同票号；
   - 剩余票数不能小于 0。

```
//ticket.c
#include <rtthread.h>

/* 静态全局变量，只在本文件内可见 */
static int remaining_tickets = 20;        /* 剩余票数 */
static int next_ticket_number = 1;        /* 下一张售出的票号 */
static rt_mutex_t ticket_mutex = RT_NULL; /* 互斥量句柄 */

/* 售票函数：返回 0 表示无票，返回正数表示售出的票号 */
int ticket_sell_one(void)
{
    int ticket_no = 0;

    if (ticket_mutex == RT_NULL) {
        rt_kprintf("Ticket system not initialized!\n");
        return 0;
    }

    /* 加锁保护完整的售票过程 */
    rt_mutex_take(ticket_mutex, RT_WAITING_FOREVER);

    if (remaining_tickets > 0) {
        ticket_no = next_ticket_number;
        next_ticket_number++;
        remaining_tickets--;
    }

    rt_mutex_release(ticket_mutex);

    return ticket_no;
}

/* 获取当前剩余票数（外部只读，不修改） */
int ticket_get_remaining(void)
{
    int count;

    rt_mutex_take(ticket_mutex, RT_WAITING_FOREVER);
    count = remaining_tickets;
    rt_mutex_release(ticket_mutex);

    return count;
}

/* 初始化票务系统（创建互斥量） */
int ticket_system_init(void)
{
    ticket_mutex = rt_mutex_create("ticket_mtx", RT_IPC_FLAG_PRIO);
    if (ticket_mutex == RT_NULL) {
        rt_kprintf("Create ticket mutex failed!\n");
        return -1;
    }
    rt_kprintf("Ticket system initialized, total: 20\n");
    return 0;
}


```

```
//ticket_thread.c
#include <rtthread.h>

/* 外部引用核心功能 */
extern int ticket_sell_one(void);
extern int ticket_get_remaining(void);

/* 窗口线程入口函数 */
static void window_thread_entry(void *parameter)
{
    int window_id = (int)(size_t)parameter;
    int ticket_no;

    while (1)
    {
        /* 尝试卖出一张票 */
        ticket_no = ticket_sell_one();

        if (ticket_no > 0) {
            /* 售票成功 */
            int remain = ticket_get_remaining();
            rt_kprintf("Window %c: Sold ticket #%d, %d left\n",
                       'A' + window_id - 1, ticket_no, remain);
        } else {
            /* 无票可卖，线程退出 */
            rt_kprintf("Window %c: No tickets left, exiting.\n",
                       'A' + window_id - 1);
            break;
        }

        /* 模拟售票处理时间（20~50ms） */
        rt_thread_mdelay(500);
    }
}

/* 创建两个售票线程（由 INIT_APP_EXPORT 或命令调用） */
int ticket_threads_start(void)
{
    rt_thread_t tid;
    int i;

    for (i = 1; i <= 2; i++) {
        char name[8];
        rt_snprintf(name, sizeof(name), "win%c", 'A' + i - 1);

        tid = rt_thread_create(name,
                               window_thread_entry,
                               (void *)(size_t)i,
                               1024,
                               20 + i,
                               10);
        if (tid != RT_NULL) {
            rt_thread_startup(tid);
            rt_kprintf("Window %c thread started.\n", 'A' + i - 1);
        } else {
            rt_kprintf("Create window %c thread failed!\n", 'A' + i - 1);
        }
    }
    return 0;
}

## 练习三：系统启动条件检查

### 场景

设备启动前需要完成三个独立模块的初始化：

- 网络模块初始化；
- 传感器模块初始化；
- 存储模块初始化。

只有三个模块全部准备完成，主业务线程才能启动。

PS：只是举例，模拟即可。

### 功能要求

定义三个事件位：

```c
#define EVENT_NET_READY      (1U << 0)
#define EVENT_SENSOR_READY   (1U << 1)
#define EVENT_STORAGE_READY  (1U << 2)
```

创建以下线程：

- 网络初始化线程；
- 传感器初始化线程；
- 存储初始化线程；
- 主业务线程。

每个初始化线程经过不同时间的延时后，发送对应的就绪事件。

主业务线程必须使用 AND 模式等待三个事件全部到达，然后打印：

```text
all modules are ready
business task started
```

另外增加一次超时测试：

- 将存储模块初始化时间设置为 5 秒；
- 主业务线程最多等待 3 秒；
- 超时后打印尚未完成的模块，并再次等待。

```
#include <rtthread.h>
#include "startup_core.h"

/* ===== 定义全局事件集对象 ===== */
rt_event_t startup_event = RT_NULL;

/* ===== 存储初始化线程（内联在 core.c 中） ===== */
void storage_init_thread_entry(void *parameter)
{
    rt_kprintf("storage init starting...\n");
    rt_thread_mdelay(5000);
    rt_event_send(startup_event, EVENT_STORAGE_READY);
    rt_kprintf("storage ready\n");
}

/* ===== 启动函数 ===== */
int startup_demo_start(void)
{
    rt_thread_t tid;

    /* 1. 创建事件集对象 */
    startup_event = rt_event_create("startup_evt", RT_IPC_FLAG_PRIO);
    if (startup_event == RT_NULL) {
        rt_kprintf("Create event failed!\n");
        return -1;
    }

    /* 2. 创建三个初始化线程 */
    tid = rt_thread_create("net_init", net_init_thread_entry, RT_NULL,
                           1024, 10, 10);
    if (tid != RT_NULL) rt_thread_startup(tid);

    tid = rt_thread_create("sensor_init", sensor_init_thread_entry, RT_NULL,
                           1024, 10, 10);
    if (tid != RT_NULL) rt_thread_startup(tid);

    tid = rt_thread_create("storage_init", storage_init_thread_entry, RT_NULL,
                           1024, 10, 10);
    if (tid != RT_NULL) rt_thread_startup(tid);

    /* 3. 创建主业务线程 */
    tid = rt_thread_create("business", business_thread_entry, RT_NULL,
                           1536, 15, 10);
    if (tid != RT_NULL) {
        rt_thread_startup(tid);
        rt_kprintf("startup_demo started\n");
        return 0;
    } else {
        rt_kprintf("Create business thread failed!\n");
        return -1;
    }
}

#include <rtthread.h>
#include "startup_core.h"

/* ===== 主业务线程入口函数 ===== */
void business_thread_entry(void *parameter)
{
    rt_uint32_t recv_events = 0;
    rt_err_t result;

    while (1)
    {
        result = rt_event_recv(startup_event,
                               EVENT_ALL_READY,
                               RT_EVENT_FLAG_AND | RT_EVENT_FLAG_CLEAR,
                               RT_TICK_PER_SECOND * 3,
                               &recv_events);

        if (result == RT_EOK) {
            rt_kprintf("all modules are ready\n");
            rt_kprintf("business task started\n");
            break;
        } else {
            rt_uint32_t current_events = 0;
            rt_event_recv(startup_event,
                          EVENT_ALL_READY,
                          RT_EVENT_FLAG_AND | RT_EVENT_FLAG_CLEAR,
                          0,
                          &current_events);

            rt_kprintf("wait modules timeout, ");
            if ((current_events & EVENT_NET_READY) == 0) {
                rt_kprintf("network ");
            }
            if ((current_events & EVENT_SENSOR_READY) == 0) {
                rt_kprintf("sensor ");
            }
            if ((current_events & EVENT_STORAGE_READY) == 0) {
                rt_kprintf("storage ");
            }
            rt_kprintf("is not ready\n");
        }
    }
}

## 练习四：快递分拣中心

### 场景

两个收件线程不断产生快递信息，一个分拣线程从消息队列中取出快递并按照目的地区域分类。

### 消息结构

```c
struct package_msg
{
    rt_uint32_t id;
    rt_uint8_t region;
    rt_uint16_t weight;
};
```

其中：

```text
region = 1：华东
region = 2：华南
region = 3：华北
```

### 功能要求

1. 创建一个消息队列，队列容量为 5 条消息。
2. 创建两个发件线程，每个线程产生 5 条快递消息。
3. 快递编号不得重复。
4. 分拣线程接收消息，并打印快递编号、区域和重量。
5. 分拣完成后统计：
   - 总快递数量；
   - 各区域快递数量；
   - 所有快递的总重量。
6. 当队列已满时，发送线程不得崩溃，应记录发送失败次数。
7. 两个发送线程完成后，向分拣线程发送一条结束消息。
8. 分拣线程收到结束消息后打印统计结果并退出。

### 示例输出

```text
package 1001 -> east, weight: 1200 g
package 2001 -> south, weight: 850 g
package 1002 -> north, weight: 960 g
...
total packages: 10
east: 4, south: 3, north: 3
total weight: 10250 g
send failed: 0

```

## 二、MSH 命令要求

每个练习导出一个独立命令：

```text

重复执行命令时，不得因同名线程或 IPC 对象残留而发生异常。可以在任务结束后主动释放资源，也可以检测任务是否仍在运行并拒绝重复启动。
